import android.accounts.AccountManager
import android.content.Intent
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import java.util.Locale
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {
    private var pending: MethodChannel.Result? = null
    private var linkCh: MethodChannel? = null
    private var voiceCh: MethodChannel? = null
    private var sr: SpeechRecognizer? = null
    private var tts: TextToSpeech? = null
    private var ttsReady = false
    private var ttsPending: String? = null
    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "ko/accounts").setMethodCallHandler { call, result ->
            if (call.method == "pick") {
                pending = result
                try {
                    val i = AccountManager.newChooseAccountIntent(null, null, arrayOf("com.google"), null, null, null, null)
                    startActivityForResult(i, 7701)
                } catch (e: Exception) { pending = null; result.success(null) }
            } else result.notImplemented()
        }
        // bolkar khojna (ko/voice): start {lang} / stop → ready, partial, final, end, error
        voiceCh = MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "ko/voice")
        voiceCh?.setMethodCallHandler { call, result ->
            when (call.method) {
                "start" -> { startVoice(call.argument<String>("lang") ?: "hi-IN"); result.success(true) }
                "stop" -> { try { sr?.stopListening() } catch (e: Exception) {}; result.success(true) }
                "say" -> { say(call.argument<String>("text") ?: ""); result.success(true) }
                else -> result.notImplemented()
            }
        }
        // dukaan ka link (karoonline://shop/N) — app khulte hi wahi dukaan
        linkCh = MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "ko/link")
        linkCh?.setMethodCallHandler { call, result ->
            if (call.method == "initial") result.success(intent?.data?.toString()) else result.notImplemented()
        }
    }
    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        val d = intent.data?.toString()
        if (d != null) linkCh?.invokeMethod("link", d)
    }
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == 7701) { pending?.success(data?.getStringExtra(AccountManager.KEY_ACCOUNT_NAME)); pending = null }
    }
    private fun startVoice(lang: String) {
        runOnUiThread {
            try {
                if (!SpeechRecognizer.isRecognitionAvailable(this)) { voiceCh?.invokeMethod("error", "na"); return@runOnUiThread }
                try { sr?.destroy() } catch (e: Exception) {}
                sr = SpeechRecognizer.createSpeechRecognizer(this)
                sr?.setRecognitionListener(object : RecognitionListener {
                    override fun onReadyForSpeech(p: Bundle?) { voiceCh?.invokeMethod("ready", "") }
                    override fun onBeginningOfSpeech() { voiceCh?.invokeMethod("begin", "") }
                    override fun onRmsChanged(v: Float) {}
                    override fun onBufferReceived(b: ByteArray?) {}
                    override fun onEndOfSpeech() { voiceCh?.invokeMethod("end", "") }
                    override fun onError(e: Int) { voiceCh?.invokeMethod("error", e.toString()) }
                    override fun onResults(r: Bundle?) {
                        val l = r?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                        voiceCh?.invokeMethod("final", l?.joinToString("|") ?: "")
                    }
                    override fun onPartialResults(r: Bundle?) {
                        val l = r?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                        if (l != null && l.isNotEmpty()) voiceCh?.invokeMethod("partial", l[0])
                    }
                    override fun onEvent(t: Int, p: Bundle?) {}
                })
                val i = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
                    putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
                    putExtra(RecognizerIntent.EXTRA_LANGUAGE, lang)
                    putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
                    putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 4)
                }
                sr?.startListening(i)
            } catch (e: Exception) { voiceCh?.invokeMethod("error", "x") }
        }
    }
    // Bolne wali awaaz (TextToSpeech, Hindi) — khatam hote hi "spoken"
    private fun speakNow(text: String) {
        try {
            tts?.language = Locale("hi", "IN")
            tts?.setSpeechRate(1.02f); tts?.setPitch(1.08f)
            tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, "ko_say")
        } catch (e: Exception) { voiceCh?.invokeMethod("spoken", "") }
    }
    private fun say(text: String) {
        runOnUiThread {
            if (tts == null) {
                ttsPending = text
                tts = TextToSpeech(this) { st ->
                    ttsReady = st == TextToSpeech.SUCCESS
                    tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
                        override fun onStart(id: String?) {}
                        override fun onDone(id: String?) { runOnUiThread { voiceCh?.invokeMethod("spoken", "") } }
                        @Deprecated("old") override fun onError(id: String?) { runOnUiThread { voiceCh?.invokeMethod("spoken", "") } }
                    })
                    val p = ttsPending; ttsPending = null
                    if (ttsReady && p != null) speakNow(p) else voiceCh?.invokeMethod("spoken", "")
                }
            } else if (ttsReady) speakNow(text) else voiceCh?.invokeMethod("spoken", "")
        }
    }
    override fun onDestroy() { try { sr?.destroy() } catch (e: Exception) {}; try { tts?.shutdown() } catch (e: Exception) {}; super.onDestroy() }
}
